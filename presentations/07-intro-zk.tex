    \subsection{Goal of the course}
    \begin{frame}{The most basic setting}
        \begin{itemize}
            \item We have a \textbf{prover} $\mathcal{P}$ and a \textbf{verifier} $\mathcal{V}$.\pause
            \item Prover $\mathcal{P}$ wants to prove some statement $\mathbbm{x}$ to the verifier.\pause
            \item Prover $\mathcal{P}$ has a \textbf{witness} $\mathbbm{w}$ that contains all necessary information to prove the statement $\mathbbm{x}$. He sends $\pi$ as a proof.\pause
            \item Verifier $\mathcal{V}$ wants to be convinced that the statement $\mathbbm{x}$ is true.
        \end{itemize}

    \begin{frame}{The Goal of SNARKs, STARKs etc.}
        We will try to solve the following problems:
        \begin{itemize}\small
            \item \textbf{Completeness:} If $\mathbbm{x}$ is true, $\pi$ proves the statement.\pause
            \item \textbf{Soundness:} If $\mathbbm{x}$ is false, the prover $\mathcal{P}$ should not be able to convince the verifier $\mathcal{V}$ via any $\pi^*$.\pause
            \item \textbf{Zero-knowledge:} $\pi$ does not reveal anything about $\mathbbm{w}$.\pause
            \item \textbf{Argument of knowledge:} Sometimes, the prover $\mathcal{P}$ should convince the verifier $\mathcal{V}$ that besides $\mathbbm{x}$ is true, he \textbf{knows} the witness $\mathbbm{w}$.\pause
            \item \textbf{Succinctness:} The proof should be short, ideally polylogarithmic in the size of the statement ($|\pi| = \mathsf{polylog}(|\mathbbm{x}|)$) + fast verification.\pause
            \item \textbf{Arithmetization:} We need to convert the statement $\mathbbm{x}$ into some algebraic form + make it relatively universal.\pause
        \end{itemize}

        \begin{remarkblock}{Note}
            SNARK, STARK, etc. will solve these problems!
        \end{remarkblock}
    \end{frame}

    \begin{frame}{Example to demonstrate the goal}
        \begin{example}
            Given a hash function $\mathsf{H}: \{0,1\}^* \to \{0,1\}^{\lambda}$, $\mathcal{P}$ wants to convince $\mathcal{V}$ that he knows the preimage $\mathbbm{w} \in \{0,1\}^*$ such that $\mathsf{H}(\mathbbm{w}) = \mathbbm{x}$.\pause
            \begin{itemize}
                \item \textbf{Zero-knowledge:} The prover $\mathcal{P}$ does not want to reveal \textit{anything} about the pre-image $\mathbbm{w}$ to the verifier $\mathcal{V}$.\pause
                \item \textbf{Argument of knowledge:} Proving $\mathbbm{x}$ has a pre-image is useless. $\mathcal{P}$ must show he \textbf{knows} $\mathbbm{w} \in \{0,1\}^*$ s.t. $\mathsf{H}(\mathbbm{w})=\mathbbm{x}$.\pause
                \item \textbf{Succinctness:} If the hash function takes $n$ operations to compute, the proof should be \textbf{much} shorter than $n$ operations. \textbf{State-of-art}: size is $\mathsf{polylog}(n) = O((\log n)^c)$. Verification time is also typically polylogarithmic (or even $O(1)$ in some cases).\pause
            \end{itemize}
        \end{example}

        \begin{remarkblock}{Note}
            But first, let us start with the basics.
        \end{remarkblock}
    \end{frame}

    \section{Relations. Languages. NP Statements.}

    \subsection{Language of true statements. Examples.}
    \begin{frame}{Language}
        \begin{definition}[Relation]
            Given two sets $\mathcal{X}$ and $\mathcal{Y}$, the \textbf{relation} is $\mathcal{R} \subseteq \mathcal{X} \times \mathcal{Y}$.\pause
            \begin{itemize}
                \item $\mathcal{X}$ is typically a set of \textbf{statements}.
                \item $\mathcal{Y}$ is a set of \textbf{witnesses}.
            \end{itemize}
        \end{definition}
        \pause

        \begin{definition}[Language of true statements]
            Let $\mathcal{R} \subseteq \mathcal{X} \times \mathcal{Y}$ be a relation. We say that a statement $\mathbbm{x} \in \mathcal{X}$ is a \textbf{true} statement if $(\mathbbm{x},\mathbbm{w}) \in \mathcal{R}$ for some $\mathbbm{w} \in \mathcal{Y}$, otherwise the statement is called \textbf{false}. We define by $\mathcal{L}_{\mathcal{R}}$ (the language over relation $\mathcal{R}$) the set of all true statements, that is:
            \begin{equation*}
                \mathcal{L}_{\mathcal{R}} = \{ \mathbbm{x} \in \mathcal{X}: \exists \mathbbm{w} \in \mathcal{Y} \; \text{such that} \; (\mathbbm{x},\mathbbm{w}) \in \mathcal{R} \}.
            \end{equation*}
        \end{definition}
        \pause

        \begin{remark}
            Further, we denote by $\mathbbm{x}$ the statement and by $\mathbbm{w}$ the witness.
        \end{remark}
    \end{frame}

    \begin{frame}{Language Example \#1: Semiprimes}
        \begin{example}[Product of Two Primes (Semiprimes)]
            \small
            \textbf{Claim:} number $n \in \mathbb{N}$ is the product of two prime numbers $\mathbbm{w} = (p,q) \in \mathbb{N} \times \mathbb{N}$. The \textbf{relation} is given by:\pause
            \begin{equation*}
                \mathcal{R} = \{ (n, p, q) \in \mathbb{N}^3: n = p \cdot q \; \text{where $p,q$ are primes} \}
            \end{equation*}
            \pause
            \vspace{-10pt}

            In this particular case, the \textbf{language of true statements} is defined as
            \begin{equation*}
                \mathcal{L}_{\mathcal{R}} = \{n \in \mathbb{N}: \exists \mathbbm{w}=(p,q) \; \text{are primes such that}\; n = p\cdot q\}
            \end{equation*}
            \pause
            \vspace{-10pt}

            \begin{itemize}
                \item \textbf{Valid witness \#1:} $n=15 \in \mathcal{L}_{\mathcal{R}}$. Witness: $\mathbbm{w}=(3,5)$.\pause
                \item \textbf{Invalid witness:} $n=16 \not\in \mathcal{L}_{\mathcal{R}}$. There is no valid witness.\pause
                \item \textbf{Valid witness \#2:} $n=50252009 \in \mathcal{L}_{\mathcal{R}}$. Witness: $\mathbbm{w}=(5749,8741)$.\pause
            \end{itemize}
        \end{example}

        \begin{questionblock}{Question}
            Is $n=27$ a true statement? What about $n=26$?
        \end{questionblock}
    \end{frame}

    \begin{frame}{Language Example \#2: Square Root}
        \begin{remarkblock}{Reminder}
            $\mathbb{Z}_N^{\times} = \{x \in \mathbb{Z}_N: \gcd(x,N)=1\}$. \textbf{Example:} $\mathbb{Z}_{10}^{\times} = \{1,3,7,9\}$.
        \end{remarkblock}
        \pause

        \begin{example}
            \textbf{Claim}: number $\mathbbm{x} \in \mathbb{Z}_N^{\times}$ is a \textbf{quadratic residue} modulo $N$: $(\exists \mathbbm{w} \in \mathbb{Z}_N^{\times}): \{\mathbbm{x} \equiv \mathbbm{w}^2 \pmod{N}\}$ ($\mathbbm{w}$ is \textbf{modular square root} of $\mathbbm{x}$).\pause

            \textbf{Relation}: $\mathcal{R} = \{ (\mathbbm{x}, \mathbbm{w}) \in (\mathbb{Z}_N^{\times})^2: \mathbbm{x} \equiv \mathbbm{w}^2 \pmod{N} \}$.\pause

            \textbf{Language:} $\mathcal{L}_{\mathcal{R}} = \{\mathbbm{x} \in \mathbb{Z}_N^{\times}: \exists \mathbbm{w} \in \mathbb{Z}_N^{\times} \; \text{such that} \; \mathbbm{x} \equiv \mathbbm{w}^2 \pmod{N}\}$.\pause

            \textbf{Examples} for $N=7$:
            \begin{itemize}
                \item $4 \in \mathcal{L}_{\mathcal{R}}$ since $5^2 \equiv 4 \pmod{7}$.\pause
                \item $3 \not\in \mathcal{L}_{\mathcal{R}}$ since there is no valid witness for $3$.\pause
            \end{itemize}
        \end{example}

        \begin{questionblock}{Question}
            Is $\mathbbm{x}=1$ a true statement for $N=5$? What about $\mathbbm{x}=4$?
        \end{questionblock}
    \end{frame}

    \begin{frame}{NP Statements}
        \begin{definition}[P Language]
            Problem is in the $\mathbf{P}$ class if exists a polytime algorithm checking $\mathbbm{x} \in \mathcal{L}$.
        \end{definition}
        \pause

        \begin{definition}[NP Language]
            A language $\mathcal{L}_{\mathcal{R}}$ belongs to the $\mathbf{NP}$ class if there exists a polynomial-time verifier $\mathcal{V}$ such that the following two properties hold:\pause
            \begin{itemize}
                \item \textbf{Completeness:} If $\mathbbm{x} \in \mathcal{L}_{\mathcal{R}}$, then there is a witness $\mathbbm{w}$ such that $\mathcal{V}(\mathbbm{x}, \mathbbm{w}) = 1$ with $|\mathbbm{w}| = \mathsf{poly}(|\mathbbm{x}|)$. Essentially, it states that true claims have \textit{short} proofs.\pause
                \item \textbf{Soundness:} If $\mathbbm{x} \not\in \mathcal{L}_{\mathcal{R}}$, then for any $\mathbbm{w}$ it holds that $\mathcal{V}(\mathbbm{x}, \mathbbm{w}) = 0$. Essentially, it states that false claims have no proofs.
            \end{itemize}
        \end{definition}
        \pause

        \begin{theorem}
            Any $\mathbf{NP}$ problem has a zero-knowledge proof (GMW86).
        \end{theorem}
    \end{frame}

    \begin{frame}{Quadratic Residue Interactive Proof: Analysis}
        \begin{block}{Interactive Protocol}
            \begin{algoen}
                \item $\mathcal{P}$ samples $r \leftarrowS \mathbb{Z}_N^{\times}$ and sends $a \gets r^2 \pmod{N}$ to $\mathcal{V}$.\pause
                \item $\mathcal{V}$ sends a random bit $b \leftarrowS \{0,1\}$ to $\mathcal{P}$.\pause
                \item $\mathcal{P}$ sends $z \gets r \cdot \mathbbm{w}^b \pmod{N}$ to $\mathcal{V}$.\pause
                \item $\mathcal{V}$ accepts if $z^2 \equiv a \cdot \mathbbm{x}^b \pmod{N}$, otherwise it rejects.\pause
                \item Repeat $\lambda \in \mathbb{N}$ times.\pause
            \end{algoen}
        \end{block}

        \begin{lemma}
            The aforementioned protocol is \textbf{complete} and \textbf{sound}.
        \end{lemma}
        \pause

        \textbf{Completeness.} If $b=0$, then $z=r$ and thus $z^2=r^2=a$, check passes.
        \pause

        If $b=1$, then $z=rw$ and thus $z^2=r^2w^2=ax$, check passes.
    \end{frame}

    \subsection{Completeness and Soundness}
    \begin{frame}{Quadratic Residue Interactive Proof: Analysis}
        \textbf{Soundness.} The main reason why the protocol is sound is inscribed in the proposition below.

        \begin{proposition}
            For any prover $\mathcal{P}^*$ with $\mathbbm{x} \not\in \mathcal{L}_{\mathcal{R}}$, the probability of $\mathcal{V}$ accepting the proof is at most $1/2$.
        \end{proposition}
        \pause

        \begin{corollary}
            After repeating the protocol $\lambda$ times, we have
            \begin{equation*}
                \Pr[\mathcal{V} \; \text{accepts after $\lambda$ rounds}] \leq \frac{1}{2^{\lambda}} = \negl(\lambda).
            \end{equation*}
        \end{corollary}

        Thus, we showed both \textbf{completeness} and \textbf{soundness} of the protocol.
    \end{frame}

    \begin{frame}{Interactive Protocol Definition}
        \textcolor{oc-gray-6}{$\langle \mathcal{P}, \mathcal{V} \rangle(\mathbbm{x})$ reads as ``interaction between $\mathcal{P}$ and $\mathcal{V}$ on the statement $\mathbbm{x}$''.}\pause

        \begin{definition}
            A pair of algorithms $(\mathcal{P},\mathcal{V})$ is called an \textbf{interactive proof} for a language $\mathcal{L}_{\mathcal{R}}$ if $\mathcal{V}$ is a polynomial-time verifier and the following two properties hold:\pause
            \begin{itemize}
                \item \textbf{Completeness:} For any $\mathbbm{x} \in \mathcal{L}_{\mathcal{R}}$, $\Pr[\langle \mathcal{P}, \mathcal{V} \rangle(\mathbbm{x}) = \mathsf{accept}]=1$.\pause
                \item \textbf{Soundness:} For any $\mathbbm{x} \not\in \mathcal{L}_{\mathcal{R}}$ and for any prover $\mathcal{P}^*$, we have
                \begin{equation*}
                    \Pr[\langle \mathcal{P}^*, \mathcal{V} \rangle(\mathbbm{x}) = \mathsf{accept}] \leq \negl(\lambda)
                \end{equation*}
            \end{itemize}
        \end{definition}
        \pause

        \begin{definition}
            \textbf{The class of interactive proofs} (\textbf{IP}) is defined as:
            \begin{equation*}
                \mathbf{IP} = \{\mathcal{L}: \text{there is an interactive proof $(\mathcal{P}, \mathcal{V})$ for $\mathcal{L}$}\}.
            \end{equation*}
        \end{definition}
    \end{frame}

    \subsection{Zero-Knowledge and Honest-Verifier Zero-Knowledge}
    \begin{frame}{Zero-Knowledge Informal Definition}
        \begin{definition}
            An interactive proof system $(\mathcal{P}, \mathcal{V})$ is called \textbf{zero-knowledge} if for any polynomial-time verifier $\mathcal{V}^*$ and any $\mathbbm{x} \in \mathcal{L}_{\mathcal{R}}$, the interaction $\langle \mathcal{P}, \mathcal{V}^* \rangle(\mathbbm{x})$ gives nothing new about the witness $\mathbbm{w}$.
        \end{definition}
        \pause

        \begin{definition}
            The pair of algorithms $(\mathcal{P}, \mathcal{V})$ is called a \textbf{zero-knowledge interactive protocol} if it is \textit{complete}, \textit{sound}, and \textit{zero-knowledge}.
        \end{definition}

            \centering
            \includegraphics[width=0.5\textwidth]{images/lecture_6/zk.pdf}
        \end{figure}
    \end{frame}

    \begin{frame}{Verifier's View}
        \begin{questionblock}{Question \#1}
            What has the verifier learned during the interaction?
        \end{questionblock}
        \pause

        \begin{itemize}
            \item First things first, he learned that the statement $\mathbbm{x}$ is true.\pause
            \item He also knows queries $(q_1,\dots,q_{\ell})$ and random coins $(r_1,\dots,r_{\ell})$ he tossed (since he is the one who has sent them).\pause
            \item Moreover, he knows the prover's messages $(m_1,m_2,\dots,m_{\ell})$.\pause
        \end{itemize}

        \begin{definition}
            All the conversation that verifier has witnessed is called \textbf{verifier's view} and is denoted as
            \begin{equation*}
                \mathsf{view}_{\mathcal{V}}(\mathcal{P}, \mathcal{V})[\mathbbm{x}] = (m_1,r_1,q_1,m_2,r_2,q_2,\dots,m_{\ell},r_{\ell},q_{\ell}).
            \end{equation*}
        \end{definition}
        \pause

        \textcolor{oc-pink-7}{\textbf{Fact:}} $\mathsf{view}_{\mathcal{V}}(\mathcal{P}, \mathcal{V})[\mathbbm{x}]$ is a \textbf{random variable}.
    \end{frame}

    \begin{frame}{Verifier's View: Example}
        \begin{example}
            For QR test, set $N:=3 \times 2^{30} + 1$ (prime number), and $\mathcal{P}$ wants to convince that $1286091780 \in \mathcal{L}_{\mathcal{R}}$. Conversation is the following:\pause
            \begin{enumerate}
                \item During the first round, $\mathcal{P}$ sends $672192003$ to $\mathcal{V}$.\pause
                \item $\mathcal{V}$ sends $b=0$ to $\mathcal{P}$.\pause
                \item $\mathcal{P}$ sends $2606437826$ to $\mathcal{V}$.\pause
                \item $\mathcal{V}$ verifies that indeed $2606437826^2 \equiv 672192003 \pmod{N}$.\pause
                \item During the second round, $\mathcal{P}$ sends $2619047580$ to $\mathcal{V}$.\pause
                \item $\mathcal{V}$ chooses $b=1$ and sends to $\mathcal{P}$.\pause
                \item $\mathcal{P}$ sends $1768388249$ to $\mathcal{V}$.\pause
                \item $\mathcal{V}$ verifies that $1768388249^2 \equiv 2619047580 \times 1286091780 \pmod{N}$.\pause
                \item Conversation ends.
            \end{enumerate}
        \end{example}
    \end{frame}

    \begin{frame}{Verifier's View: Example}
        \begin{example}
            The \textbf{view of the verifier} $\mathcal{V}$ is the following:
            \begin{align*}
                &\mathsf{view}_{\mathcal{V}}(\mathcal{P}, \mathcal{V})[1286091780] \\
                &\quad= (672192003, 0, 2606437826, 2619047580, 1, 1768388249)
            \end{align*}
            \pause

            \begin{itemize}
                \item Essentially, this view is the same as you have witnessed.\pause
                \item You have not learned anything about $\mathbbm{w}$ that prover $\mathcal{P}$ knows.\pause
                \item The witness was $\mathbbm{w} = 3042517305$ and two randomnesses were $r_1 = 2606437826$ and $r_2 = 3023142760$.\pause
                \item This is a random variable: conversation could be different.
            \end{itemize}
        \end{example}
    \end{frame}

    \begin{frame}{Zero-Knowledge Formally: Simulation Paradigm}
        \begin{questionblock}{Question \#2}
            What does it mean that the protocol is zero-knowledge?
        \end{questionblock}
        \pause

        \begin{itemize}
            \item Protocol is zero-knowledge if, given the verifier's $\mathsf{view}_{\mathcal{V}}(\mathcal{P}, \mathcal{V})[\mathbbm{x}]$, verifier cannot infer any information about the witness $\mathbbm{w}$.\pause
            \item What does it mean that verifier $\mathcal{V}$ learns nothing new? It means that this view could have been simulated by $\mathcal{V}$ \textit{without even running an interaction}.\pause
            \item Call the view after the real interaction as \textbf{real view}, while the view after the simulation as \textbf{simulated view}.\pause
        \end{itemize}

        \begin{remarkblock}{Note}
            Such idea of defining the zero-knowledge is called \textbf{simulation paradigm} and currently the most widely used way to prove zero-knowledge.
        \end{remarkblock}
    \end{frame}

    \begin{frame}{Computational Indistinguishability}
        \begin{figure}
            \centering
            \includegraphics[width=0.8\textwidth]{images/lecture_6/cind.pdf}
        \end{figure}
        \pause

        \begin{definition}[Informal Computational Indistinguishability]
            $D_0$ and $D_1$ are \textbf{computationally indistinguishable} (denoted by $D_0 \approx D_1$) if for any PPT distinguisher $\mathcal{A}$, even after polynomial number $k$ of samples from $D_b$ (where $b \leftarrowS \{0,1\}$), for prediction $\widehat{b}$: $\Pr[\widehat{b} = b] < \frac{1}{2} + \negl(k)$.
        \end{definition}
    \end{frame}

    \begin{frame}{Zero-Knowledge Formally (Kind of)}
        Finally, we are ready to define the \textbf{zero-knowledge}.

        \begin{definition}[Honest-Verifier Zero-Knowledge (HVZK)]
            An interactive protocol $(\mathcal{P}, \mathcal{V})$ is \textbf{honest-verifier zero-knowledge (HVZK)} for a language $\mathcal{L}_{\mathcal{R}}$ if there exists a poly-time simulator $\mathsf{Sim}$ such that for any valid statement $\mathbbm{x} \in \mathcal{L}_{\mathcal{R}}$:
            \begin{equation*}
                \mathsf{view}_{\mathcal{V}}(\mathcal{P}, \mathcal{V})[\mathbbm{x}] \approx \mathsf{Sim}(\mathbbm{x}, \textcolor{oc-gray-6}{1^{\lambda}})
            \end{equation*}
        \end{definition}
        \pause

        \begin{definition}[Zero-Knowledge (ZK)]
            An interactive protocol $(\mathcal{P}, \mathcal{V})$ is \textbf{zero-knowledge (ZK)} for a language $\mathcal{L}_{\mathcal{R}}$ if \textit{for every poly-time verifier} $\mathcal{V}^*$ there \textit{exists a poly-time simulator} $\mathsf{Sim}$ such that for any valid statement $\mathbbm{x} \in \mathcal{L}_{\mathcal{R}}$:
            \begin{equation*}
                \mathsf{view}_{\mathcal{V}^*}(\mathcal{P}, \mathcal{V}^*)[\mathbbm{x}] \approx \mathsf{Sim}(\mathbbm{x}, \textcolor{oc-gray-6}{1^{\lambda}})
            \end{equation*}
        \end{definition}
    \end{frame}

    \subsection{Proof of Knowledge}

    \begin{frame}{Proof of Knowledge: Why?}
        Now, the main issue with the above definition is that \textit{we have proven the statement correctness, but we have not proven that the prover \textbf{knows} the witness}. These are completely two distinct things!\pause

        \begin{example}
            Consider the \textbf{discrete logarithm relation and language} for a cyclic group $E(\mathbb{F}_p)$ of order $r$:
            \begin{align*}
                \mathcal{R} = \{(P, \alpha) \in E(\mathbb{F}_p) \times \mathbb{Z}_r: P = [\alpha] G\}, \\ \mathcal{L}_{\mathcal{R}} = \{P \in E(\mathbb{F}_p): \exists \alpha \in \mathbb{Z}_r \; \text{such that} \; P = [\alpha] G\}
            \end{align*}
        \end{example}
        \pause

        \begin{questionblock}{Question}
            What does it mean that $P \in \mathcal{L}_{\mathcal{R}}$?
        \end{questionblock}
        \pause

        Turns out $\mathcal{L}_{\mathcal{R}} = E(\mathbb{F}_p)$, so the proof $P \in \mathcal{L}_{\mathcal{R}}$ itself is useless.
    \end{frame}

    \begin{frame}{Proof of Knowledge: Definition}
        \begin{enumerate}
            \item The knowledge of witness means that we can \textbf{extract} the witness while interacting with the prover.\pause
            \item Thus, there should be an algorithm called \textbf{extractor} $\mathcal{E}$ which can extract the witness $\mathbbm{w}$.\pause
            \item $\mathcal{E}$ is given more power than $\mathcal{V}$ (otherwise, if the protocol is zero-knowledge, we cannot extract $\mathbbm{w}$). $\mathcal{E}$ can \textbf{rewind} and \textbf{call} prover $\mathcal{P}$ multiple times.\pause
            \item Sometimes, this is referred to as ``extractor $\mathcal{E}$ uses $\mathcal{P}$ as an oracle''.\pause
        \end{enumerate}

        \begin{definition}[Proof of Knowledge]
            The interactive protocol $(\mathcal{P}, \mathcal{V})$ is a \textbf{proof of knowledge} for $\mathcal{L}_{\mathcal{R}}$ if there exists a poly-time extractor algorithm $\mathcal{E}$ such that for any valid statement $\mathbbm{x} \in \mathcal{L}_{\mathcal{R}}$, in expected poly-time $\mathcal{E}^{\mathcal{P}}(\mathbbm{x})$ outputs $\mathbbm{w}$ such that $(\mathbbm{x},\mathbbm{w}) \in \mathcal{R}$.
        \end{definition}
    \end{frame}

    \begin{frame}{Proof of Knowledge: Example}
        \begin{lemma}
            The quadratic residue interactive protocol is a proof of knowledge.
        \end{lemma}
        \pause

        \textbf{Proof.} Let us define the extractor $\mathcal{E}$ for the statement $\mathbbm{x}$ as follows:
        \begin{algoen}
            \item Run the prover to receive $a \equiv r^2 \pmod{N}$ ($r$ is chosen randomly from $\mathbb{Z}_N^{\times}$).\pause
            \item Set verifier's message to $b=0$ to get $z_1 \gets r$.\pause
            \item \textbf{Rewind} and set verifier's message to $b=1$ to get $z_2 \gets rw \pmod{N}$.\pause
            \item Output $z_2/z_1 \pmod{N}$.\pause
        \end{algoen}

        The extractor $\mathcal{E}$ will always output $\mathbbm{w}$ if $\mathbbm{x} \in \mathcal{L}_{\mathcal{R}}$. $\hfill\Box$
    \end{frame}

    \section{Fiat-Shamir Heuristic}
    \subsection{Cryptographic Oracles}
    \begin{frame}{Cryptographic Oracle}
        \begin{definition}[Cryptographic Oracle]
            Informally, \textit{cryptographic oracle} is simply a function $\mathcal{O}$ that gives in $O(1)$ an answer to some typically very hard problem.
        \end{definition}
        \pause

        \begin{example}[CDH Problem]
            Consider the \textbf{Computational Diffie-Hellman (CDH)} problem on the cyclic elliptic curve $E(\mathbb{F}_p)$ of prime order $r$ with a generator $G$.\pause

            \textbf{Hard Problem:} $[\alpha\beta]G$ given $[\alpha]G$ and $[\beta]G$ where $\alpha,\beta \in \mathbb{Z}_r$.\pause

            \textbf{Oracle:} However, we \textit{could} assume that such problem can be solved in $O(1)$ by a cryptographic oracle $\mathcal{O}_{\text{CDH}}: ([\alpha]G,[\beta]G) \mapsto [\alpha\beta]G$.\pause

            This way, we can rigorously prove the security of some cryptographic protocols \textit{even} if the Diffie-Hellman problem is suddenly solved.
        \end{example}
    \end{frame}

    \begin{frame}{Random Oracle (RO)}
        One of the most popular cryptographic oracles is the \textbf{random oracle} $\mathcal{O}_{\text{R}}$.

        \begin{definition}[Informal definition of RO]
            Suppose someone is inputting $x$ to the random oracle $\mathcal{O}_{\text{R}}: \mathcal{X} \to \mathcal{Y}$\footnote{Typically, RO works with a family of functions $f: \mathcal{X} \to \mathcal{Y}$, but we are not going too deep into the details.}. The oracle $\mathcal{O}_{\text{R}}$ does the following:\pause
            \begin{enumerate}
                \item If $x$ has been queried before, the oracle returns the same value as it returned before.\pause
                \item If $x$ has not been queried before, the oracle returns a randomly uniformly sampled value from the output space $\mathcal{Y}$.\pause
            \end{enumerate}
        \end{definition}

        \begin{questionblock}{Question}
            Which very well-known cryptographic object can ``serve'' as a random oracle?
        \end{questionblock}
    \end{frame}

    \subsection{Fiat-Shamir Transformation}
    \begin{frame}{Fiat-Shamir Transformation}
        \begin{block}{Statement}
            \textbf{Any} interactive public-coin protocol can be converted into a non-interactive public-coin protocol with preserving completeness, soundness, and zero-knowledge using the random oracle.
        \end{block}

        One of such transformations is called \textbf{Fiat-Shamir heuristic}. Idea:\pause
        \begin{enumerate}
            \item If all what $\mathcal{V}$ does is sending uniformly random values, this is an overkill.\pause
            \item Instead of $\mathcal{V}$ sending random values, prover should be able to generate it himself, but he should not know the randomness in advance.\pause
            \item Thus, we can replace the verifier's messages with the hash $\mathsf{H}$ (random oracle) of all the previous conversation.
        \end{enumerate}
    \end{frame}
